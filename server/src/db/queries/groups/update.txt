//QUERY 1: Group owner changes the group name


EXPLAIN ANALYZE
UPDATE "Groups"
SET "groupName" = 'Computer Science Students'
WHERE "groupID" = 42
  AND "userID" = 7;



//QUERY 2: Group owner changes the group description


EXPLAIN ANALYZE
UPDATE "Groups"
SET "description" = 'A group for students to discuss computer science, projects, and events.'
WHERE "groupID" = 42
  AND "userID" = 7;



//QUERY 3: Group owner transfers ownership to another user


EXPLAIN ANALYZE
UPDATE "Groups"
SET "userID" = 15
WHERE "groupID" = 42
  AND "userID" = 7;



//QUERY 4: Admin edits a group announcement


EXPLAIN ANALYZE
UPDATE "Announcements"
SET "announcementTitle" = 'Updated Meeting Announcement',
    "content" = 'The group meeting has been moved to Friday at 3 PM.'
WHERE "announcementID" = 101
  AND "groupID" = 42;



//QUERY 5: Admin edits a group event


EXPLAIN ANALYZE
UPDATE "Events"
SET "eventTitle" = 'Computer Science Study Session',
    "content" = 'The study session will now be held in Room 204.'
WHERE "eventID" = 25
  AND "groupID" = 42;



//QUERY 6: Member edits their comment in the group


EXPLAIN ANALYZE
UPDATE "Comments"
SET "commentContent" = 'Updated comment about the group discussion.'
WHERE "commentID" = 125
  AND "userID" = 15
  AND "isDeleted" = FALSE;



//QUERY 7: Member deletes their comment using soft deletion


EXPLAIN ANALYZE
UPDATE "Comments"
SET "isDeleted" = TRUE,
    "timeDeleted" = NOW()
WHERE "commentID" = 125
  AND "userID" = 15
  AND "isDeleted" = FALSE;



//QUERY 8: Admin approves a request to join the group


EXPLAIN ANALYZE
UPDATE "PermissionRequest"
SET "status" = 'APPROVED',
    "reviewedAt" = NOW()
WHERE "requestID" = 315
  AND "groupID" = 42
  AND "status" = 'PENDING';



//QUERY 9: Admin rejects a request to join the group


EXPLAIN ANALYZE
UPDATE "PermissionRequest"
SET "status" = 'REJECTED',
    "reviewedAt" = NOW()
WHERE "requestID" = 315
  AND "groupID" = 42
  AND "status" = 'PENDING';
