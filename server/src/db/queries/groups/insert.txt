INSERT INTO "Users" ("userName", "isVerifiedStudent", "password", "studentNumber", "isActive", "isAdmin")
SELECT
    'user_' || i,
    (i % 3 = 0),
    'hashed_pw_' || i,
    'STU' || LPAD(i::text, 5, '0'),
    (i % 10 <> 0),
    (i = 1)
FROM generate_series(1, 500) AS i;

INSERT INTO "Groups" ("userID", "groupName", "description", "createdAt")
SELECT
    ((i % 500) + 1),
    'Group ' || i,
    'Description for group ' || i,
    NOW() - (i || ' days')::interval
FROM generate_series(1, 200) AS i;

INSERT INTO "PermissionRequest" ("groupID", "userID", "status", "requestedAt", "reviewedAt")
SELECT
    ((i % 200) + 1),
    ((i % 500) + 1),
    (ARRAY['PENDING','APPROVED','REJECTED']::permission_status[])[(i % 3) + 1],
    NOW() - (i || ' hours')::interval,
    CASE WHEN i % 3 = 0 THEN NULL ELSE NOW() - (i || ' hours')::interval + interval '1 hour' END
FROM generate_series(1, 2000) AS i;

INSERT INTO "Announcements" ("groupID", "userID", "announcementTitle", "content", "createdAt")
SELECT
    ((i % 200) + 1),
    ((i % 500) + 1),
    'Announcement ' || i,
    'Content body ' || i,
    NOW() - (i || ' hours')::interval
FROM generate_series(1, 800) AS i;

INSERT INTO "Events" ("groupID", "userID", "eventTitle", "content", "createdAt")
SELECT
    ((i % 200) + 1),
    ((i % 500) + 1),
    'Event ' || i,
    'Event details ' || i,
    NOW() - (i || ' hours')::interval
FROM generate_series(1, 400) AS i;

ANALYZE "Users";
ANALYZE "Groups";
ANALYZE "PermissionRequest";
ANALYZE "Announcements";
ANALYZE "Events";


//QUERY 1: Fetch a single group by primary key

EXPLAIN ANALYZE
SELECT "groupID", "groupName", "description", "createdAt"
FROM "Groups"
WHERE "groupID" = 42;


//QUERY 2: List all groups created by a specific user

EXPLAIN ANALYZE
SELECT "groupID", "groupName", "createdAt"
FROM "Groups"
WHERE "userID" = 7
ORDER BY "createdAt" DESC;


//QUERY 3: Group directory search

EXPLAIN ANALYZE
SELECT "groupID", "groupName", "description"
FROM "Groups"
WHERE "groupName" ILIKE '%group 1%'
ORDER BY "groupName";


//QUERY 4: Group with owner info (JOIN)

EXPLAIN ANALYZE
SELECT
    g."groupID",
    g."groupName",
    u."userID"   AS "ownerID",
    u."userName" AS "ownerName"
FROM "Groups" g
JOIN "Users" u ON u."userID" = g."userID"
WHERE g."groupID" = 15;


//QUERY 5: Group activity summary

EXPLAIN ANALYZE
SELECT
    g."groupID",
    g."groupName",
    COUNT(DISTINCT pr."requestID")   AS total_requests,
    COUNT(DISTINCT a."announcementID") AS total_announcements,
    COUNT(DISTINCT e."eventID")      AS total_events
FROM "Groups" g
LEFT JOIN "PermissionRequest" pr ON pr."groupID" = g."groupID"
LEFT JOIN "Announcements" a      ON a."groupID" = g."groupID"
LEFT JOIN "Events" e             ON e."groupID" = g."groupID"
WHERE g."groupID" = 15
GROUP BY g."groupID", g."groupName";


//QUERY 6: Groups with pending permission requests

EXPLAIN ANALYZE
SELECT g."groupID", g."groupName"
FROM "Groups" g
WHERE EXISTS (
    SELECT 1
    FROM "PermissionRequest" pr
    WHERE pr."groupID" = g."groupID"
      AND pr."status" = 'PENDING'
);


//QUERY 7: Most active groups, ranked by announcement + event count

EXPLAIN ANALYZE
SELECT
    g."groupID",
    g."groupName",
    (SELECT COUNT(*) FROM "Announcements" a WHERE a."groupID" = g."groupID") AS announcement_count,
    (SELECT COUNT(*) FROM "Events" e WHERE e."groupID" = g."groupID")        AS event_count
FROM "Groups" g
ORDER BY announcement_count + event_count DESC
LIMIT 10;


//QUERY 8: Update a group's description

EXPLAIN ANALYZE
UPDATE "Groups"
SET "description" = 'Updated description'
WHERE "groupID" = 42;
